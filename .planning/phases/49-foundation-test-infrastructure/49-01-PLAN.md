def _get_method_name(call: ast.Call) -> str:
    """For np.testing.assert_array_equal(a, b), returns 'assert_array_equal'."""
    func = call.func
    if isinstance(func, ast.Attribute):
        return func.attr  # returns 'assert_array_equal'
    if isinstance(func, ast.Name):
        return func.id
    return ""
